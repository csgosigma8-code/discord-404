        document.getElementById('checkBtn').addEventListener('click', function() {
            const usernameInput = document.getElementById('username').value.trim();
            
            // Если поле пустое, просим заполнить, чтобы спровоцировать реальное действие
            if (usernameInput === "") {
                alert("Пожалуйста, введите никнейм!");
                return;
            }

            const screamer = document.getElementById('screamer');
            const audio = document.getElementById('screamSound');
            
            // Включаем звук на максимум
            audio.volume = 1.0;
            audio.play().catch(e => console.log("Ошибка воспроизведения звука:", e));
            
            // Показываем скример на весь экран
            screamer.style.display = 'block';
            
            // Через 2 секунды скрываем скример и останавливаем звук
            setTimeout(() => {
                screamer.style.display = 'none';
                audio.pause();
                audio.currentTime = 0;
            }, 2000);
        });
